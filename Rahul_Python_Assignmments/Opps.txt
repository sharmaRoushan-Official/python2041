# Demonstrate *multilevel inheritance* using classes A → B → C, 
# each with a method. Call all three methods from an object of class C.

class A:
    def __init__(self, name):
        self.name = name

    def name_val(self):
        print(f"Name: {self.name}")


class B(A):
    def __init__(self, name, age):
        super().__init__(name)
        self.age = age

    def age_val(self):
        print(f"Age: {self.age}")


class C(B):
    def __init__(self, name, age, salary):
        super().__init__(name, age)
        self.salary = salary

    def sal_val(self):
        print(f"Salary: {self.salary}")


a = C("Rahul", 32, 100)

a.name_val()
a.age_val()
a.sal_val()

class Shape(ABC):
   @abstractmethod
   def area(self): 
        pass
    
   @abstractmethod
   def perimeter(self): 
       pass

    def area(self):
        print("Area of circle...")

    def perimeter(self):
        print("Perimeter of circle...")

class Square(Shape):
    def area(self):
        print("Area of square...")

    def perimeter(self):
        print("Perimeter of square...")
